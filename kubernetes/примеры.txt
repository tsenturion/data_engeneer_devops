apiVersion: v1
kind: Pod
metadata:
  name: frontend-pod
  lables:
    app: frontend
spec:
  containers:
    - name: nginx
      image: nginx:latest
  
  affinity:
    podAntiAffinity:
      preferredDuringSchedulingIgnoredDuringExecution:
        - labelSelector:
            matchExpressions:
              - key: app
                operator: In
                values:
                  - frontend
          topologyKey: kubernetes.io/hostname
